export function triageLoad(
  signals: Signal[],
  queueDepth: number
): string {
  let score = 0;
  for (const s of signals) {
    if (s.kind === 'error')
      score += s.count * 6;
  }
  return score > 80 ? 'hot' : 'ok';
}
